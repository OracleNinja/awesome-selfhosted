# KDP Studio dependencies.
#
# The core — models, print geometry, gates, pipeline, assembly, CLI — needs
# NOTHING from this file. It is stdlib-only by design, including the PNG
# decoder and the PDF writer, so a gate can always be evaluated and a verdict
# is never skipped because an install failed. CI has a job that proves it.
#
# The packages below unlock the checks that genuinely cannot be done without
# them. Until they are installed those gates report BLOCKED, which stops a book
# rather than passing it. That is the intended behaviour, not a limitation.

# --- Testing (all the core suite needs) --------------------------------------
pytest==9.1.1

# --- Pixel inspection: line-art QA (G9) and near-duplicate detection (G4) ----
# Stray shading, unclosed regions, watermarks, mirrored repeats.
pillow==12.3.0
imagehash==4.3.2

# --- PDF inspection: interior QA (G10) and cover QA (G11) -------------------
# Note this is a *reader*. PDFs are written by kdp/pdfwrite.py using only the
# standard library, so the thing that builds a book is not the thing that
# certifies it — a bug in the writer cannot pass itself.
pypdf==6.1.1
